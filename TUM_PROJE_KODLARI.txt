package com.shoppingapp.data.dao

import androidx.room.*
import com.shoppingapp.data.model.*
import kotlinx.coroutines.flow.Flow

@Dao
interface ShoppingDao {
    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertProduct(product: Product): Long

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertStore(store: Store): Long

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertCategory(category: Category): Long

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertListItem(item: ListItem): Long

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertPantryItem(item: PantryItem): Long

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertPurchaseHistory(history: PurchaseHistory): Long

    @Query("SELECT * FROM products WHERE barcode = :barcode")
    suspend fun getProductByBarcode(barcode: String): Product?

    @Query("SELECT * FROM products WHERE id = :id")
    suspend fun getProductById(id: Long): Product?

    @Query("SELECT * FROM products")
    fun getAllProducts(): Flow<List<Product>>

    @Query("SELECT * FROM list_items")
    fun getAllListItems(): Flow<List<ListItem>>

    @Query("SELECT * FROM list_items WHERE isPurchased = 0")
    fun getActiveListItemsFlow(): Flow<List<ListItem>>

    @Query("SELECT * FROM stores")
    fun getAllStores(): Flow<List<Store>>

    @Query("SELECT * FROM categories")
    fun getAllCategories(): Flow<List<Category>>

    @Query("SELECT * FROM pantry_items")
    fun getPantryItems(): Flow<List<PantryItem>>

    @Query("SELECT * FROM purchase_history")
    fun getPurchaseHistory(): Flow<List<PurchaseHistory>>

    @Update
    suspend fun updateListItem(item: ListItem)

    @Delete
    suspend fun deleteListItem(item: ListItem)

    @Query("DELETE FROM list_items WHERE isPurchased = 1")
    suspend fun clearPurchasedItems()
}

import android.content.Context

import dagger.hilt.android.qualifiers.ApplicationContext
import dagger.hilt.components.SingletonComponent
import javax.inject.Singleton

@Module
@InstallIn(SingletonComponent::class)
object DatabaseModule {

    @Provides
    @Singleton
    fun provideDatabase(@ApplicationContext context: Context): ShoppingDatabase {
        return Room.databaseBuilder(
            context,

        return database.shoppingDao()
    }

package com.shoppingapp.feature.stats

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.shoppingapp.data.repository.ShoppingRepository
import dagger.hilt.android.lifecycle.HiltViewModel
import kotlinx.coroutines.flow.*
import javax.inject.Inject

data class StatsUiState(
    val totalSpending: Double = 0.0,
    val pfandTotal: Double = 0.0,
    val activeItemCount: Int = 0,
    val purchasedItemCount: Int = 0
)

@HiltViewModel
class StatsViewModel @Inject constructor(
    private val repository: ShoppingRepository
) : ViewModel() {

    val statsState: StateFlow<StatsUiState> = combine(
        repository.allListItems,
        repository.allProducts,
        repository.purchaseHistory
    ) { items, products, history ->
        val productMap = products.associateBy { it.id }

        var currentPfand = 0.0
        var activeCount = 0
        var purchasedCount = 0

        items.forEach { item ->
            val product = productMap[item.productId]
            if (product != null && product.pfandAmount > 0) {
                currentPfand += product.pfandAmount * item.quantity
            }
            if (item.isPurchased) {
                purchasedCount++
            } else {
                activeCount++
            }
        }

        val historySpending = history.sumOf { it.totalAmount }
        val historyPfand = history.sumOf { it.pfandTotal }

        StatsUiState(
            totalSpending = historySpending,
            pfandTotal = historyPfand + currentPfand,
            activeItemCount = activeCount,
            purchasedItemCount = purchasedCount
        )
    }.stateIn(
        scope = viewModelScope,
        started = SharingStarted.WhileSubscribed(5000),
        initialValue = StatsUiState()
    )
}

class PfandViewModel @Inject constructor() : ViewModel() {
    private val _uiState = MutableStateFlow(PfandUiState())
    val uiState: StateFlow<PfandUiState> = _uiState

    fun updateSmallCount(value: String) {
        _uiState.update { it.copy(smallCount = value) }
    }

    fun updateLargeCount(value: String) {
        _uiState.update { it.copy(largeCount = value) }
    }

    fun updateTinCount(value: String) {
        _uiState.update { it.copy(tinCount = value) }
    }

    fun calculateDailyEarnings() {
        val small = _uiState.value.smallCount.toIntOrNull() ?: 0
        val large = _uiState.value.largeCount.toIntOrNull() ?: 0
        val tin = _uiState.value.tinCount.toIntOrNull() ?: 0
        val daily = small * 0.15 + large * 0.25 + tin * 0.25
        _uiState.update {
            it.copy(
                dailyEarnings = "${"%.2f".format(daily)}",
                totalEarnings = "${"%.2f".format((it.totalEarnings.toDoubleOrNull() ?: 0.0) + daily}"
            )
        }
    }

    fun resetCounts() {
        _uiState.update {
            it.copy(
                smallCount = "",
                largeCount = "",
                tinCount = "",
                dailyEarnings = "0.00"
            )
        }
    }
}

package com.shoppingapp.ui

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.shoppingapp.data.model.*
import com.shoppingapp.data.repository.ShoppingRepository
import dagger.hilt.android.lifecycle.HiltViewModel
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch
import javax.inject.Inject

data class UiShoppingItem(
    val listItem: ListItem,
    val product: Product,
    val store: Store?,
    val category: Category?
)

data class ShoppingUiState(
    val items: List<UiShoppingItem> = emptyList(),
    val stores: List<Store> = emptyList(),
    val categories: List<Category> = emptyList(),
    val selectedStoreId: Long? = null,
    val isLoading: Boolean = true
)

@HiltViewModel
class ShoppingViewModel @Inject constructor(
    private val repository: ShoppingRepository
) : ViewModel() {

    private val _selectedStoreId = MutableStateFlow<Long?>(null)

    val uiState: StateFlow<ShoppingUiState> = combine(
        repository.allListItems,
        repository.allProducts,
        repository.allStores,
        repository.allCategories,
        _selectedStoreId
    ) { items, products, stores, categories, selectedStore ->
        val productMap = products.associateBy { it.id }
        val storeMap = stores.associateBy { it.id }
        val categoryMap = categories.associateBy { it.id }

        val uiItems = items.mapNotNull { item ->
            val product = productMap[item.productId] ?: return@mapNotNull null
            val store = storeMap[item.storeId]
            val category = categoryMap[product.categoryId]
            UiShoppingItem(
                listItem = item,
                product = product,
                store = store,
                category = category
            )
        }.filter {
            selectedStore == null || it.listItem.storeId == selectedStore
        }

        ShoppingUiState(
            items = uiItems,
            stores = stores,
            categories = categories,
            selectedStoreId = selectedStore,
            isLoading = false
        )
    }.stateIn(
        scope = viewModelScope,
        started = SharingStarted.WhileSubscribed(5000),
        initialValue = ShoppingUiState()
    )

    init {
        viewModelScope.launch {
            repository.initDefaultData()
        }
    }

    fun selectStoreFilter(storeId: Long?) {
        _selectedStoreId.value = storeId
    }

    fun togglePurchased(item: ListItem) {
        viewModelScope.launch {
            repository.updateListItem(item.copy(isPurchased = !item.isPurchased))
        }
    }

    fun deleteItem(item: ListItem) {
        viewModelScope.launch {
            repository.deleteListItem(item)
        }
    }

    fun addNewItem(
        name: String,
        storeId: Long,
        categoryId: Long,
        quantity: Double,
        unit: String,
        pfandAmount: Double
    ) {
        viewModelScope.launch {
            val product = Product(
                id = System.currentTimeMillis(),
                name = name,
                barcode = "CUSTOM_${System.currentTimeMillis()}",
                categoryId = categoryId,
                unit = unit,
                defaultStoreId = storeId,
                pfandAmount = pfandAmount
            )
            repository.addProduct(product)
            repository.addListItem(productId = product.id, storeId = storeId, quantity = quantity)
        }
    }

    fun clearPurchased() {
        viewModelScope.launch {
            repository.clearPurchasedItems()
        }
    }
}
